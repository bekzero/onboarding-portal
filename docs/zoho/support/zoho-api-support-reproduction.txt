C. OAUTH IMPLEMENTATION
=======================

The server uses the OAuth refresh-token grant. It sends the client ID, client secret,
grant type, and refresh token as application/x-www-form-urlencoded data to the US Accounts
endpoint. The returned access token is cached in server memory until 60 seconds before
expiry. A 401 response invalidates the cache and causes one forced refresh/retry.

The original refresh response reported these actual granted scopes:

- ZohoMarketingAutomation.lead.READ
- ZohoMarketingAutomation.lead.UPDATE
- ZohoMarketingAutomation.journey.READ

This scope value came from the OAuth refresh response, not merely from local configuration.
A second independently generated refresh token also produced INVALID_OAUTHSCOPE for Get
Mailing Lists, but the second token's actual granted scopes have not been independently
verified. The OAuth refresh response did not provide a Marketing Automation organization ID.

Exact sanitized excerpt from lib/zoho-api-core.ts (credentials remain runtime values):

----- BEGIN CURRENT OAUTH EXCERPT -----
export const ZOHO_API_BASE_URL = "https://marketingautomation.zoho.com";
export const ZOHO_ACCOUNTS_BASE_URL = "https://accounts.zoho.com";
export const ZOHO_TOPICS_API_BASE_URL = "https://ma.zoho.com";
export const ZOHO_ORGANIZATION_ID = "org859403413";

async getAccessToken(forceRefresh = false) {
  if (!forceRefresh && this.cache && this.cache.expiresAt - 60_000 > this.now()) {
    return this.cache.accessToken;
  }
  if (this.refreshPromise) return this.refreshPromise;

  this.refreshPromise = this.refreshAccessToken();
  try {
    return await this.refreshPromise;
  } finally {
    this.refreshPromise = null;
  }
}

private async refreshAccessToken() {
  requireOAuthConfig(this.config);
  const body = new URLSearchParams({
    client_id: this.config.clientId,
    client_secret: this.config.clientSecret,
    grant_type: "refresh_token",
    refresh_token: this.config.refreshToken
  });
  const response = await fetchWithTimeout(
    this.fetchImpl,
    `${ZOHO_ACCOUNTS_BASE_URL}/oauth/v2/token`,
    {
      body,
      headers: { "content-type": "application/x-www-form-urlencoded" },
      method: "POST"
    },
    this.timeoutMs
  );
  const payload = await parseJsonResponse(response);
  const record = isRecord(payload) ? payload : {};
  const accessToken = typeof record.access_token === "string" ? record.access_token : "";
  const expiresIn = Number(record.expires_in ?? 3600);

  if (!response.ok || !accessToken) {
    throw new ZohoApiError(
      sanitizeProviderMessage(record.error_description ?? record.error ?? "Zoho OAuth refresh failed."),
      "OAUTH",
      { status: response.status }
    );
  }

  this.cache = {
    accessToken,
    expiresAt: this.now() + (Number.isFinite(expiresIn) ? expiresIn : 3600) * 1000
  };
  this.lastRefreshDiagnostic = {
    apiDomain: typeof record.api_domain === "string" ? record.api_domain : null,
    grantedScope: typeof record.scope === "string" ? record.scope : null,
    responseKeys: Object.keys(record)
      .filter((key) => !/token|secret|authorization/i.test(key))
      .sort(),
    tokenType: typeof record.token_type === "string" ? record.token_type : null
  };
  return accessToken;
}

----- BEGIN CURRENT MAILING-LIST EXCERPT -----
async inspectMailingListsPage(input: { fromIndex: number; range: number }) {
  const body = new URLSearchParams({
    fromindex: String(input.fromIndex),
    range: String(input.range),
    resfmt: "JSON",
    sort: "asc"
  });
  const result = await this.requestResult(
    "/api/v1/getmailinglists",
    {
      body,
      headers: { "content-type": "application/x-www-form-urlencoded" },
      method: "POST"
    },
    true,
    ZOHO_API_BASE_URL,
    true
  );
  const diagnostic = buildMailingListsPageDiagnostic(result.payload, result.status);
  const code = providerCode(result.payload);
  if (code && !["0", "200"].includes(code)) {
    const providerError = classifyProviderError(
      new Response(null, { status: result.status }),
      result.payload
    );
    return {
      diagnostic,
      lists: null,
      parseError: { kind: providerError.kind, message: providerError.message }
    };
  }
  try {
    return { diagnostic, lists: parseMailingLists(result.payload), parseError: null };
  } catch (error) {
    return {
      diagnostic,
      lists: null,
      parseError: error instanceof ZohoApiError
        ? { kind: error.kind, message: error.message }
        : { kind: "UNEXPECTED", message: "The mailing-list response could not be parsed." }
    };
  }
}

----- BEGIN CURRENT TOPICS EXCERPT -----
async getTopicsPage(input: { baseUrl?: string; fromIndex: number; range: number }) {
  const details = encodeURIComponent(`{from_index:${input.fromIndex},range:${input.range}}`);
  const result = await this.requestResult(
    `/api/v1/topics?details=${details}`,
    {},
    true,
    input.baseUrl ?? ZOHO_TOPICS_API_BASE_URL
  );
  return {
    diagnostic: buildTopicsPageDiagnostic(result.payload, result.status),
    topics: parseTopics(result.payload)
  };
}

async discoverTopics(input: { baseUrl?: string; maxPages?: number; pageSize?: number } = {}) {
  const maxPages = Math.min(Math.max(input.maxPages ?? 10, 1), 10);
  const pageSize = Math.min(Math.max(input.pageSize ?? 100, 1), 100);
  const pages: ZohoTopicsPageDiagnostic[] = [];
  const topics = new Map<string, ZohoTopic>();

  for (let page = 0; page < maxPages; page += 1) {
    const result = await this.getTopicsPage({
      baseUrl: input.baseUrl,
      fromIndex: page * pageSize,
      range: pageSize
    });
    pages.push(result.diagnostic);
    result.topics.forEach((topic) => topics.set(topic.topicId, topic));
    if (result.topics.length < pageSize) {
      return { pages, topics: [...topics.values()] };
    }
  }

  throw new ZohoApiError(
    "Zoho topic discovery exceeded the bounded pagination limit.",
    "INVALID_RESPONSE"
  );
}

J. STANDALONE READ-ONLY NODE.JS REPRODUCTION
============================================

The following is a newly created standalone reproduction for Zoho Support. It is not a
verbatim application file. It mirrors the current request methods, endpoints, parameters,
authorization scheme, bounded timeout, and sanitized output. It requires Node.js 18 or later.

Save only this section as zoho-readonly-repro.mjs if execution is required. Set environment
variables in the process without placing secrets in source control. The script refuses to run
unless the expected organization string and explicit read-only authorization are present.
That local check is a safeguard; it does not independently prove OAuth portal binding.

----- BEGIN STANDALONE REPRODUCTION -----
const EXPECTED_ORGANIZATION = "org859403413";
const ACCOUNTS_BASE = "https://accounts.zoho.com";
const MA_BASE = "https://marketingautomation.zoho.com";
const TOPICS_BASES = ["https://ma.zoho.com", MA_BASE];

function required(name) {
  const value = process.env[name]?.trim();
  if (!value) throw new Error(`Missing required environment variable: ${name}`);
  return value;
}

if (required("EXPECTED_ZOHO_ORGANIZATION") !== EXPECTED_ORGANIZATION) {
  throw new Error("Expected Zoho organization does not match the approved KZero value.");
}
if (required("ALLOW_ZOHO_READ_ONLY_REPRO") !== "I_AUTHORIZE_READ_ONLY_ZOHO_REPRO") {
  throw new Error("Explicit read-only authorization is missing.");
}

const oauth = {
  clientId: required("ZOHO_CLIENT_ID"),
  clientSecret: required("ZOHO_CLIENT_SECRET"),
  refreshToken: required("ZOHO_REFRESH_TOKEN")
};

async function request(url, init = {}, timeoutMs = 10000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    return await fetch(url, { ...init, signal: controller.signal });
  } finally {
    clearTimeout(timer);
  }
}

async function json(response) {
  const text = await response.text();
  try {
    return JSON.parse(text);
  } catch {
    return { diagnostic: "NON_JSON_RESPONSE", responseBytes: text.length };
  }
}

async function refreshAccessToken() {
  const body = new URLSearchParams({
    client_id: oauth.clientId,
    client_secret: oauth.clientSecret,
    grant_type: "refresh_token",
    refresh_token: oauth.refreshToken
  });
  const response = await request(`${ACCOUNTS_BASE}/oauth/v2/token`, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body
  });
  const payload = await json(response);
  if (!response.ok || typeof payload.access_token !== "string") {
    throw new Error(`OAuth refresh failed (HTTP ${response.status}, error=${String(payload.error ?? "unknown")})`);
  }
  return {
    accessToken: payload.access_token,
    diagnostic: {
      httpStatus: response.status,
      apiDomain: typeof payload.api_domain === "string" ? payload.api_domain : null,
      grantedScope: typeof payload.scope === "string" ? payload.scope : null,
      tokenType: typeof payload.token_type === "string" ? payload.token_type : null
    }
  };
}

function sanitizedDiagnostic(response, payload) {
  const source = payload && typeof payload === "object" ? payload : {};
  const topicDetails = Array.isArray(source.topicDetails) ? source.topicDetails : null;
  const lists = Array.isArray(source.list_of_details)
    ? source.list_of_details
    : Array.isArray(source.lists) ? source.lists : null;
  return {
    httpStatus: response.status,
    applicationStatus: typeof source.status === "string" ? source.status : null,
    applicationCode: source.code === undefined && source.Code === undefined
      ? null : String(source.code ?? source.Code),
    message: typeof (source.message ?? source.Message) === "string"
      ? String(source.message ?? source.Message).slice(0, 300) : null,
    topLevelKeys: Object.keys(source).filter((key) => !/token|secret|authorization/i.test(key)).sort(),
    topicCount: topicDetails?.length ?? null,
    mailingListCount: lists?.length ?? null
  };
}

async function authenticatedRequest(accessToken, url, init = {}) {
  const response = await request(url, {
    ...init,
    headers: {
      ...init.headers,
      Authorization: `Zoho-oauthtoken ${accessToken}`
    }
  });
  const payload = await json(response);
  return sanitizedDiagnostic(response, payload);
}

const { accessToken, diagnostic: oauthDiagnostic } = await refreshAccessToken();

const mailingListBody = new URLSearchParams({
  fromindex: "1",
  range: "100",
  resfmt: "JSON",
  sort: "asc"
});
const mailingLists = await authenticatedRequest(
  accessToken,
  `${MA_BASE}/api/v1/getmailinglists`,
  {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body: mailingListBody
  }
);

const details = encodeURIComponent("{from_index:0,range:100}");
const topics = [];
for (const base of TOPICS_BASES) {
  topics.push({
    base,
    result: await authenticatedRequest(accessToken, `${base}/api/v1/topics?details=${details}`)
  });
}

console.log(JSON.stringify({
  environment: {
    expectedOrganization: EXPECTED_ORGANIZATION,
    organizationBindingVerifiedByResponse: false,
    dataCenter: "US"
  },
  oauth: oauthDiagnostic,
  mailingLists,
  topics
}, null, 2));
